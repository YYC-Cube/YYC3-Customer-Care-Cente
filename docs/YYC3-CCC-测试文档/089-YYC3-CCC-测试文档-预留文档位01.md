##### 环境切换

```typescript
import { beforeAll, afterAll } from 'vitest';

export class TestEnvironmentManager {
  private static currentEnv: TestEnvironment | null = null;

  static async setupEnvironment(envName: string) {
    this.currentEnv = getTestEnvironment(envName);

    process.env.API_BASE_URL = this.currentEnv.url;
    process.env.DATABASE_URL = this.currentEnv.database;
    process.env.REDIS_URL = this.currentEnv.redis;

    console.log(`✅ 切换到测试环境: ${envName}`);
  }

  static async cleanupEnvironment() {
    if (this.currentEnv) {
      await this.cleanupDatabase();
      await this.cleanupRedis();

      this.currentEnv = null;
      console.log('✅ 清理测试环境完成');
    }
  }

  private static async cleanupDatabase() {
    const prisma = new PrismaClient({
      datasources: {
        db: { url: process.env.DATABASE_URL },
      },
    });

    await prisma.user.deleteMany();
    await prisma.service.deleteMany();
    await prisma.order.deleteMany();

    await prisma.$disconnect();
  }

  private static async cleanupRedis() {
    const redis = createClient({ url: process.env.REDIS_URL });
    await redis.connect();
    await redis.flushDb();
    await redis.disconnect();
  }
}

beforeAll(async () => {
  await TestEnvironmentManager.setupEnvironment('local');
});

afterAll(async () => {
  await TestEnvironmentManager.cleanupEnvironment();
});
```

#### 4.2 测试性能基准

##### 性能指标

```typescript
export interface PerformanceMetrics {
  name: string;
  duration: number;
  memory: number;
  cpu: number;
  timestamp: number;
}

export class PerformanceBenchmark {
  private static metrics: PerformanceMetrics[] = [];

  static async measure<T>(name: string, fn: () => Promise<T>): Promise<T> {
    const startTime = performance.now();
    const startMemory = process.memoryUsage().heapUsed;

    const result = await fn();

    const endTime = performance.now();
    const endMemory = process.memoryUsage().heapUsed;

    const metrics: PerformanceMetrics = {
      name,
      duration: endTime - startTime,
      memory: endMemory - startMemory,
      cpu: process.cpuUsage().user,
      timestamp: Date.now(),
    };

    this.metrics.push(metrics);

    console.log(`⏱️ ${name}: ${metrics.duration.toFixed(2)}ms`);

    return result;
  }

  static getMetrics(): PerformanceMetrics[] {
    return this.metrics;
  }

  static getAverageMetrics(name: string): PerformanceMetrics | null {
    const filtered = this.metrics.filter((m) => m.name === name);
    if (filtered.length === 0) return null;

    return {
      name,
      duration:
        filtered.reduce((sum, m) => sum + m.duration, 0) / filtered.length,
      memory: filtered.reduce((sum, m) => sum + m.memory, 0) / filtered.length,
      cpu: filtered.reduce((sum, m) => sum + m.cpu, 0) / filtered.length,
      timestamp: Date.now(),
    };
  }

  static clearMetrics() {
    this.metrics = [];
  }
}
```

##### 性能测试示例

```typescript
describe('性能测试', () => {
  it('API响应时间应该小于200ms', async () => {
    const response = await PerformanceBenchmark.measure(
      'GET /api/services',
      () => fetch('http://localhost:3200/api/services')
    );

    const metrics = PerformanceBenchmark.getAverageMetrics('GET /api/services');
    expect(metrics?.duration).toBeLessThan(200);
  });

  it('批量创建用户应该在1秒内完成', async () => {
    const users = Array.from({ length: 100 }, () =>
      TestDataGenerator.generateUser()
    );

    await PerformanceBenchmark.measure('批量创建用户', () =>
      prisma.user.createMany({ data: users })
    );

    const metrics = PerformanceBenchmark.getAverageMetrics('批量创建用户');
    expect(metrics?.duration).toBeLessThan(1000);
  });
});
```

#### 4.4 测试报告生成

##### HTML报告生成

```typescript
import { writeFileSync } from 'fs';
import { join } from 'path';

export class TestReportGenerator {
  static generateHTMLReport(results: TestResult[]): string {
    const passed = results.filter((r) => r.status === 'passed').length;
    const failed = results.filter((r) => r.status === 'failed').length;
    const skipped = results.filter((r) => r.status === 'skipped').length;
    const total = results.length;

    return `
<!DOCTYPE html>
<html>
<head>
  <title>测试报告</title>
  <style>
    body { font-family: Arial, sans-serif; margin: 20px; }
    .summary { display: flex; gap: 20px; margin-bottom: 20px; }
    .card { padding: 20px; border-radius: 8px; flex: 1; }
    .passed { background: #4caf50; color: white; }
    .failed { background: #f44336; color: white; }
    .skipped { background: #ff9800; color: white; }
    table { width: 100%; border-collapse: collapse; }
    th, td { padding: 10px; text-align: left; border-bottom: 1px solid #ddd; }
  </style>
</head>
<body>
  <h1>测试报告</h1>
  <div class="summary">
    <div class="card passed">通过: ${passed}</div>
    <div class="card failed">失败: ${failed}</div>
    <div class="card skipped">跳过: ${skipped}</div>
    <div class="card">总计: ${total}</div>
  </div>
  <table>
    <thead>
      <tr>
        <th>测试名称</th>
        <th>状态</th>
        <th>耗时</th>
        <th>错误信息</th>
      </tr>
    </thead>
    <tbody>
      ${results
        .map(
          (r) => `
        <tr>
          <td>${r.name}</td>
          <td>${r.status}</td>
          <td>${r.duration}ms</td>
          <td>${r.error || ''}</td>
        </tr>
      `
        )
        .join('')}
    </tbody>
  </table>
</body>
</html>
    `;
  }

  static saveReport(report: string, filename: string) {
    const path = join(process.cwd(), 'coverage', filename);
    writeFileSync(path, report);
    console.log(`✅ 报告已保存: ${path}`);
  }
}
```

#### 4.5 测试最佳实践

##### 测试金字塔

```
      /\
     /E2E\      10%
    /------\
   /集成测试\    30%
  /----------\
 /  单元测试  \  60%
/--------------\
```

##### 测试原则

1. **独立性**：每个测试应该独立运行，不依赖其他测试
2. **可重复性**：测试结果应该一致，不受外部因素影响
3. **快速性**：单元测试应该在毫秒级别完成
4. **清晰性**：测试名称应该清晰描述测试意图
5. **维护性**：测试代码应该易于理解和维护

##### 测试反模式

- ❌ 测试实现细节而非行为
- ❌ 过度使用Mock导致测试脆弱
- ❌ 测试之间共享状态
- ❌ 测试代码复杂度过高
- ❌ 忽略测试覆盖率
